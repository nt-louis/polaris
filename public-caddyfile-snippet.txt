# ==============================================================================
# PUBLIC HOST CADDYFILE SNIPPET (TEMPLATE / EXAMPLE)
#
# This file contains example reverse proxy rules for your public-facing
# Host Caddy (the one running directly on your VPS host).
# Because we are using the Nested Gateway Pattern with Tailscale,
# the Host Caddy routes traffic across the Tailnet directly to the
# isolated Gateway nodes using Tailnet FQDNs (e.g. core.<tailnet>.ts.net)
# without needing internal port mappings exposed on the host.
#
# NOTE: All site blocks use the http:// scheme because TLS is terminated
# upstream by Cloudflare Tunnels (or external reverse proxy). This prevents
# Caddy from attempting to auto-provision HTTPS certificates in a tunnel setup.
#
# REVIEW RULE: This is a reusable template, not a generated inventory. Before
# copying a site block, verify that its upstream exists in active compose
# discovery and that its gateway/path is listed in Docs/TAILSCALE_URLS.md.
# Several historical examples below intentionally remain for migration
# reference, including Emby, Readarr, Kavita, Coolify, and Zurg-related routes.
#
# NetBird is an exception: its gRPC/control-plane deployment requires native
# HTTPS and HTTP/2 support. Do not route NetBird through a Cloudflare HTTP/1.1
# tunnel; use the dedicated native HTTPS configuration described in the
# NetBird runbooks.
# ==============================================================================

{
    # Global Options
    servers {
        protocols h1 h2c
        trusted_proxies static private_ranges
    }
}

# ==============================================================================
# Snippets (Reusable Configuration Blocks)
# ==============================================================================

# Basic Security Headers
(base_security) {
    header {
        X-Frame-Options "SAMEORIGIN"
        X-Content-Type-Options "nosniff"
        X-XSS-Protection "1; mode=block"
        Referrer-Policy "strict-origin-when-cross-origin"
        -Server # Hide Server version for security
    }
}

# CORS Snippet for Stremio Addons
(cors) {
    @options method OPTIONS
    handle @options {
        header Access-Control-Allow-Origin "*"
        header Access-Control-Allow-Methods "GET, POST, OPTIONS"
        header Access-Control-Allow-Headers "*"
        header Access-Control-Max-Age "86400"
        respond "" 204
    }
    header Access-Control-Allow-Origin "*"
    header Access-Control-Allow-Headers "*"
}

# Proxy Settings (Pass Real IPs to Containers)
(proxy_settings) {
    header_up Host {host}
    header_up X-Real-IP {remote_host}
    header_up X-Forwarded-Proto https
}

# oauth2-proxy Forward Auth Snippet
(oauth2-proxy) {
    forward_auth 127.0.0.1:4180 {
        uri /oauth2/auth
        copy_headers X-Auth-Request-Preferred-Username>X-WebAuth-User X-Auth-Request-Email X-Auth-Request-User

        @error status 401
        handle_response @error {
            redir * /oauth2/start?rd=https://{host}{uri}
        }
    }
}

# Reusable OIDC Protection Snippet
# Usage: import protected <upstream-target>
(protected) {
    handle /oauth2/* {
        reverse_proxy 127.0.0.1:4180
    }
    handle {
        import oauth2-proxy
        reverse_proxy {args[0]} {
            import proxy_settings
        }
    }
}
